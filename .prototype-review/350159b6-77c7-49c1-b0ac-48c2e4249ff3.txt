<script type="text/x-dc" data-dc-script>
class Component extends DCLogic {
  state = (() => {
    const q = new URLSearchParams(location.search), h = location.hash.slice(1);
    const role = (q.get('role') || (h.includes('realtor') ? 'realtor' : 'customer'));
    return { role: role === 'realtor' ? 'realtor' : 'customer', cust: q.get('as') === 'owner' ? 'owner' : 'seeker',
      mode: (q.get('mode') || (h.includes('signup') ? 'signup' : 'signin')) === 'signup' ? 'signup' : 'signin',
      step: 'phone', phone: '', otp: '', fname: '', lname: '', email: '' };
  })();
  renderVals() {
    const s = this.state, set = o => this.setState(o);
    const realtor = s.role === 'realtor', owner = !realtor && s.cust === 'owner', signup = s.mode === 'signup';
    const reset = { step: 'phone', otp: '' };
    const C = {realtor:{
      eyebrow: 'FOR REALTORS', headline: 'Run your real estate business.',
      sub: '',
      points: [['settings', 'AI agents that actually work as a team'], ['file-text', 'List a property in under a minute'], ['play-circle', 'Branded videos, brochures, flyers and posts with a tap'], ['share', 'Ready to upload for every social platform'], ['user', 'Leads that manage themselves'], ['message', 'Co-broking matches with verified realtors'], ['pie-chart', 'Your whole business at a glance']]
    }, owner:{
      eyebrow: 'FOR PROPERTY OWNERS', headline: 'List it. Manage it. Close it.',
      sub: '',
      points: [['file-text', 'List in minutes with a live preview'], ['bar-chart', 'Price analysis from real transactions'], ['user', 'Realtor assistance, only if you choose']]
    }, seeker:{
      eyebrow: 'FOR PROPERTY SEEKERS', headline: 'Genuine listings only.',
      sub: '', bold: 'Only available. No fakes. No duplicates.',
      points: [['shield-check', 'Verified listings and RERA-registered projects'], ['search', 'Search properties your way'], ['bell-off', 'No spam calls. Realtors reach you only when you ask.'], ['star', 'Save properties and track visits in one place']]
    }};
    const keys = realtor ? ['realtor'] : ['seeker','owner'];
    const okP = s.phone.length === 10, okO = s.otp.length === 4, okN = s.fname.trim() && s.lname.trim();
    const okR = !realtor || ((s.firm || '').trim() && (s.rera || '').length >= 8 && s.rcity && /.+@.+\..+/.test(s.email || ''));
    const ready = s.step === 'phone' ? (okP && (!signup || (okN && okR))) : okO;
    const suOpts = realtor ? ['Realtor', 'Realtor firm'] : ['Property seeker', 'Property owner'];
    const done = s.step === 'done';
    return {
      tagline: realtor ? 'Powering India\u2019s Leading Realtors' : 'Real Estate Simplified!',
      roleTabs: [['customer', 'Customer'], ['realtor', 'Realtor']].map(([id, label]) => { const on = s.role === id; return { label, bg: on ? '#fff' : 'transparent', shadow: on ? 'var(--shadow-sm)' : 'none', fg: on ? '#000' : 'var(--rez-graphite)', weight: on ? 700 : 500, onClick: () => set({ role: id, ...reset }) }; }),
      isCustomer: !realtor,
      blocks: keys.map((k, i) => ({ ...C[k], points: C[k].points.map(([icon, t]) => ({ icon, t })), size: realtor ? 'clamp(30px,3.4vw,40px)' : 'clamp(28px,3vw,36px)', pb: i < keys.length - 1 ? '28px' : '0', bb: i < keys.length - 1 ? '1px solid var(--rez-steel)' : '0' })),
      cardTitle: done ? (signup ? 'You\u2019re in.' : 'Welcome back.') : s.step === 'otp' ? 'Enter the OTP.' : signup ? 'Sign up' : 'Sign in',
      cardSub: s.step === 'otp' ? 'We\u2019ve sent a 4-digit code to your mobile.' : signup ? 'Create your Rezide account.' : (realtor ? 'Log in to your realtor dashboard.' : 'Log in with your mobile number.'),
      isSignup: signup && !done, isSignupForm: signup && !realtor && s.step === 'phone', isRealtorSignup: signup && realtor && s.step === 'phone',
      firm: s.firm || '', rera: s.rera || '', rcity: s.rcity || '', assoc: s.assoc || '',
      onFirm: e => set({ firm: e.target.value }), onRera: e => set({ rera: e.target.value.toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 14) }),
      onRcity: e => set({ rcity: e.target.value }), onAssoc: e => set({ assoc: e.target.value }),
      rcityFg: s.rcity ? '#000' : '#8A8F99', assocFg: s.assoc ? '#000' : '#8A8F99',
      cityOpts: ['Mumbai', 'Pune', 'Bengaluru', 'Delhi NCR', 'Hyderabad', 'Chennai', 'Ahmedabad', 'Kolkata'],
      assocOpts: ['NAR-India Independent', 'NAR-India, AREA', 'NAR-India, SMART', 'NAR-India, TREK', 'Other association'],
      suRole: realtor ? (s.suRole || 'Realtor') : (owner ? 'Property owner' : 'Property seeker'), suRoleOpts: suOpts,
      onSuRole: e => { const v = e.target.value; realtor ? set({ suRole: v }) : set({ cust: v === 'Property owner' ? 'owner' : 'seeker' }); },
      fname: s.fname, lname: s.lname, email: s.email,
      onFname: e => set({ fname: e.target.value }), onLname: e => set({ lname: e.target.value }), onEmail: e => set({ email: e.target.value }),
      showPhone: s.step === 'phone', showOtp: s.step === 'otp', isDone: done,
      phone: s.phone, otp: s.otp,
      onPhone: e => set({ phone: e.target.value.replace(/\D/g, '').slice(0, 10) }),
      onOtp: e => set({ otp: e.target.value.replace(/\D/g, '').slice(0, 4) }),
      editPhone: () => set(reset), resend: e => { e.preventDefault(); set({ otp: '' }); },
      showBtn: !done, btnLabel: s.step === 'otp' ? (signup ? (realtor ? 'Verify & Start 14-Day Trial' : 'Verify & Sign Up') : 'Verify & Sign In') : 'Send OTP',
      btnBg: ready ? '#000' : 'var(--rez-steel)',
      onAction: () => { if (!ready) return;
        if (s.step === 'otp' && !realtor) { try { localStorage.setItem('rz_cust', JSON.stringify({ role: owner ? 'owner' : 'seeker', fname: s.fname || 'Viraj', lname: s.lname || 'Bhatia', phone: s.phone })); } catch (e) {} window.location.href = '#account'; return; }
        if (s.step === 'otp' && realtor) { window.location.href = '#dashboard'; return; }
        set({ step: s.step === 'phone' ? 'otp' : 'done' }); },
      doneMsg: realtor && signup ? 'Your 14-day trial has started. We\u2019ll verify your RERA ID within 24 hours.' : realtor ? 'Your realtor dashboard is ready.' : owner ? 'You can now list and manage your property.' : 'Your saved homes and visits are ready.',
      doneCta: realtor ? 'Go to Realtor Dashboard' : owner ? 'Go to My Listings' : 'Go to My Account',
      doneHref: realtor ? '#dashboard' : '#account',
      switchQ: signup ? 'Already have an account?' : 'Don\u2019t have an account?',
      switchLabel: signup ? 'Sign in' : 'Sign up',
      switchMode: () => set({ mode: signup ? 'signin' : 'signup', ...reset })
    };
  }
}

const fmtINR=n=>n>=1e7?'₹ '+(n/1e7).toFixed(2)+' Cr':n>=1e5?'₹ '+(n/1e5).toFixed(2)+' L':'₹ '+Math.round(n/1e3)+'k';
const fmtN=n=>n.toLocaleString('en-IN');
function makeListings(st,count=9,offset=0){
  const c=CAT[st.type], rent=st.mode==='rent', f=st.lf||{}, out=[];
  for(let k=0;k<count;k++){
    const i=offset+k;
    const h=hash(st.city+st.type+st.mode+JSON.stringify(f)+st.locality+i), h2=hash('x'+h);
    const fu=pickA(f.unit,h), fs=pickA(f.size,h), fe=pickA(f.extra,h);
    const loc=f.loc||LOC[st.city][(h+i)%8];
    const name=(st.type==='homes'&&f.project)||NAMES[h%NAMES.length]+' '+SUFFIX[h2%SUFFIX.length];
    const kind=fu||c.kinds[h%c.kinds.length];
    let area, specs, title;
    if(st.type==='homes'){
      const bhk=fs||c.size[1][h%4]; const b=parseInt(bhk)||5;
      area=450+b*320+h%300; title=String.fromCharCode(65+h%4)+' - '+(100*(1+h%22)+1+h2%6)+', '+name;
      specs=[{v:bhk,l:kind.replace(/s$/,'')},{v:Math.max(1,b-(h%2))+' Baths',l:'Bathrooms'},{v:(1+h%2)+' Parks',l:'Parking'},{v:(fe||['Fully furnished','Semi furnished','Unfurnished'][h%3]).replace(/ furnished/i,''),l:'Furnishing'}];
    } else if(st.type==='land'){
      area=2400+h*7%40000; title='Plot '+(10+h%180)+', '+NAMES[h%NAMES.length]+' Enclave';
      specs=[{v:['East','North','West','North-East'][h%4],l:'Facing'},{v:(20+h%5*10)+' ft',l:'Road width'},{v:kind.split(' ')[0],l:'Zone'},{v:(fe||['NA approved','RERA approved','Title clear'][h%3]).split(' ')[0],l:'Approval'}];
    } else {
      area=st.type==='industrial'?8000+h*5%60000:600+h*3%9000;
      title=(st.type==='retail'?'Shop ':'Unit ')+(1+h%9)+'0'+(1+h2%9)+', '+name;
      specs=st.type==='offices'?[{v:Math.round(area/90)+'',l:'Seats'},{v:(1+h%6)+'',l:'Cabins'},{v:(2+h%10)+'',l:'Parking'},{v:(fe||['Furnished','Warm shell','Bare shell'][h%3]).split(' ')[0],l:'Fit-out'}]
        :st.type==='retail'?[{v:(14+h%30)+' ft',l:'Frontage'},{v:(fe||['Ground floor','Upper floor','Mezzanine'][h%3]).split(' ')[0],l:'Floor'},{v:h%2?'Private':'Shared',l:'Washroom'},{v:(1+h%6)+'',l:'Parking'}]
        :[{v:(24+h%20)+' ft',l:'Clear height'},{v:(1+h%8)+'',l:'Docks'},{v:(fe||['Up to 100 kVA','100 – 500 kVA','500 kVA+'][h%3]).replace('Up to ','≤'),l:'Power'},{v:(4+h%20)+'',l:'Parking'}];
    }
    const rate=rent?(st.type==='homes'?40+h%110:60+h%200):(st.type==='land'?4000+h%20000:12000+h%55000);
    const total=area*rate;
    out.push({title,loc:loc+', '+st.city,price:fmtINR(total)+(rent?' / mo':''),rate:'₹ '+fmtN(rate)+(rent?' per sq ft / mo':' per sq ft'),area:(st.type==='land'?'Plot: ':st.type==='industrial'?'Built-up: ':'Carpet: ')+fmtN(area)+' sq ft',
      specs,k1:specs[0].v,k2:specs[1].v,keyline:specs[0].v+' · '+specs[1].v+' · '+fmtN(area)+' sq ft',areaShort:fmtN(area)+' sq ft',_loc:loc,realtor:REALTORS[h%REALTORS.length],updated:MONTHS[h2%MONTHS.length],badge:h%3?'Verified':'Available',slot:'lst-'+st.type+'-'+i,ph:kind+' photo',_p:total,_a:area});
  }
  if(st.sort==='low') out.sort((a,b)=>a._p-b._p); else if(st.sort==='high') out.sort((a,b)=>b._p-a._p); else if(st.sort==='area') out.sort((a,b)=>b._a-a._a);
  return out;
}
